    // ── Apply Grayscale Filter ──────────────────────────────────
    function _edApplyGrayscale() {
      _edEnsureEditsPx(); _edSaveUndo();
      const pw = _edCvsEditsPx.width, ph = _edCvsEditsPx.height;
      const ctx = _edCvsEditsPx.getContext("2d", { willReadFrequently: true });
      const imgData = ctx.getImageData(0, 0, pw, ph);
      const d = imgData.data;

      let lMskData = null;
      if (typeof edLassoOps !== 'undefined' && edLassoOps.length > 0) {
          const lMsk = buildLassoMaskCanvas(pw, ph);
          lMskData = lMsk.getContext("2d", {willReadFrequently: true}).getImageData(0, 0, pw, ph).data;
      }
      
      for (let i = 0; i < d.length; i += 4) {
          let apply = true;
          if (lMskData) {
               apply = (lMskData[i + 3] > 0);
          }
          if (apply) {
              const avg = d[i] * 0.299 + d[i+1] * 0.587 + d[i+2] * 0.114;
              d[i] = avg; d[i+1] = avg; d[i+2] = avg;
          }
      }
      ctx.putImageData(imgData, 0, 0);
      redraw();
    }

    // ── Apply Gaussian Blur Filter ──────────────────────────────
    function _edApplyGaussianBlur(radius) {
      radius = radius || 5;
      _edEnsureEditsPx(); _edSaveUndo();
      const pw = _edCvsEditsPx.width, ph = _edCvsEditsPx.height;
      const srcCtx = _edCvsEditsPx.getContext("2d", { willReadFrequently: true });

      // Check lasso mask
      let lMskCvs = null;
      const hasLasso = typeof edLassoOps !== 'undefined' && edLassoOps.length > 0;
      if (hasLasso) {
        lMskCvs = buildLassoMaskCanvas(pw, ph);
      }

      // Create temp canvas for blur
      const tmpCvs = document.createElement("canvas");
      tmpCvs.width = pw; tmpCvs.height = ph;
      const tmpCtx = tmpCvs.getContext("2d");
      tmpCtx.filter = `blur(${radius}px)`;
      tmpCtx.drawImage(_edCvsEditsPx, 0, 0);
      tmpCtx.filter = "none";

      if (hasLasso && lMskCvs) {
        // Composite: only apply blur inside lasso selection
        const orig = srcCtx.getImageData(0, 0, pw, ph);
        const blurred = tmpCtx.getImageData(0, 0, pw, ph);
        const mask = lMskCvs.getContext("2d", {willReadFrequently:true}).getImageData(0, 0, pw, ph).data;
        const od = orig.data, bd = blurred.data;
        for (let i = 0; i < od.length; i += 4) {
          const a = mask[i + 3] / 255; // mask alpha as blend factor
          od[i]     = od[i]     * (1 - a) + bd[i]     * a;
          od[i + 1] = od[i + 1] * (1 - a) + bd[i + 1] * a;
          od[i + 2] = od[i + 2] * (1 - a) + bd[i + 2] * a;
        }
        srcCtx.putImageData(orig, 0, 0);
      } else {
        // Full image blur
        srcCtx.clearRect(0, 0, pw, ph);
        srcCtx.drawImage(tmpCvs, 0, 0);
      }
      redraw();
    }

    // ── Apply Levels Correction ─────────────────────────────────
    function _edApplyLevels(shadows, gamma, highlights) {
      shadows = shadows || 0;
      gamma = gamma || 1;
      highlights = (highlights != null) ? highlights : 255;
      if (shadows >= highlights) return;
      _edEnsureEditsPx(); _edSaveUndo();
      const pw = _edCvsEditsPx.width, ph = _edCvsEditsPx.height;
      const ctx = _edCvsEditsPx.getContext("2d", { willReadFrequently: true });
      const imgData = ctx.getImageData(0, 0, pw, ph);
      const d = imgData.data;
      const invGamma = 1 / gamma;
      const range = highlights - shadows;

      let lMskData = null;
      if (typeof edLassoOps !== 'undefined' && edLassoOps.length > 0) {
        const lMsk = buildLassoMaskCanvas(pw, ph);
        lMskData = lMsk.getContext("2d", {willReadFrequently: true}).getImageData(0, 0, pw, ph).data;
      }

      for (let i = 0; i < d.length; i += 4) {
        if (lMskData && lMskData[i + 3] === 0) continue;
        for (let c = 0; c < 3; c++) {
          let v = d[i + c];
          v = Math.max(0, Math.min(1, (v - shadows) / range));
          v = Math.pow(v, invGamma);
          d[i + c] = Math.round(v * 255);
        }
      }
      ctx.putImageData(imgData, 0, 0);
      redraw();
    }

    function _syncPixelToolUI() {
      Object.entries(ptBtns).forEach(([k, b]) => {
        const on = edPixelTool === k;

        b.style.borderColor = on ? "#445599" : "#3a3a3a";
      });
      ptColorWrapper.style.display = (edPixelTool === "brush" || edPixelTool === "eyedropper") ? "flex" : "none";
      ptAlphaRow.style.display = (edPixelTool === "brush") ? "flex" : "none";

        ptSmVal.textContent = ptSmSlider.value + "%";

        ptSmSlider.value = String(Math.round(_edBlurStr * 100));
        ptSmVal.textContent = ptSmSlider.value + "%";
      }

      ptBrushRow.style.display = needsSize ? "flex" : "none";

    function _selectPixelTool(t) {
      if (edPixelTool === t) { edPixelTool = null; } // toggle off
      else {
        edPixelTool = t;
        // Mutual exclusion: disable crop only (lasso stays active in pixels mode for CA Fill)
        if (edCropMode) { edCropMode = false; syncCropToggle(); }
        if (typeof edLassoMode !== 'undefined' && edLassoMode) { edLassoMode = false; edLassoDrawing = false; edLassoCurrentPts = []; stopLassoAnts(); syncLassoToggle(); }
        _edEnsureEditsPx();
      }
      _syncPixelToolUI();
      if (!edPixelTool && !edCropMode && !edLassoMode) {
        ca.style.cursor = "grab"; hint.textContent = "Drag to pan · Scroll to zoom";
      }
      redraw();
    }
